[settings]
# HOW OFTEN: after each automatic event the next one comes after a random time between min and max game
# days (decimals are fine, e.g. 0.5-1.5).  0-0 = that kind never happens by itself.  Only when the panel switch
# "automatic events" is on and somebody is online.
airdrop_days = 6-14
heli_days = 10-22
airdrop_chance = 100             # % chance that an automatic airdrop happens when its timer ends (0-100); otherwise the timer starts again
heli_chance = 100                # the same for the automatic helicopter crash
horde_chance = 100               # the same for an automatic horde (a town whose cooldown ended); on "no" the town's cooldown starts again
# HORDES: every town has its OWN cooldown (see [cities], the last number: days min-max).  When a town's cooldown is over
# and a player is online, a horde walks to a spot in that town (any town may be the target, near players or not: the
# waves wait until somebody comes close enough for the forest to load).  After the attack the town's cooldown starts anew.
# horde_city_days is used for a town without its own value.  The panel switch "automatic hordes" turns all of this off.
horde_city_days = 14-20
horde_max_active = 1             # at most this many automatic hordes at the same time (towns that are due wait their turn)

# AIRDROP: automatic radio warnings in the last day before the drop (hours before: every 2 hours).
# The warnings reach EVERY powered-on receiver on EVERY channel.  The position of the landed crate is
# broadcast after the drop on the wave of this event (see radio_frequencies below) and is heard by every
# receiver tuned to it.
airdrop_warn_hours = 24 22 20 18 16 14 12 10 8 6 4 2
airdrop_repeat_hours = 1         # after the drop, every N hours: EVERY channel hears "the cargo was already dropped, coordinates on wave F"; the coordinates only on that wave ...
airdrop_repeat_count = 4         # ... this many times (4 x 1 h)
airdrop_crates = 1
airdrop_loot = random           # random  or  set:NAME   (e.g. set:Humanitarian cargo)

# HELICOPTER, in order:
#  1. heli_sos_lead_hours before the crash: a signal caught "by chance" (broken, with static): the call number
#     of the helicopter, the crew talking to the command about other states, the pilot says they will enter
#     the Kentucky airspace in half an hour.  Every receiver, every channel.
#  2. at the crash: SOS with the wave for the rescue group (no coordinates).  Every channel.
#  3. then every heli_sos_repeat_hours, heli_sos_repeat_count times (12 = 12 hours): SOS with the exact
#     coordinates on the rescue wave of this crash - heard by receivers tuned to it.
heli_sos_lead_hours = 0.5
heli_sos_repeat_hours = 1
heli_sos_repeat_count = 12
# RADIO WAVES (kHz).  Every airdrop, helicopter crash and horde gets ONE of these at random; the radio texts
# say which ({F}).  For the helicopter this is the wave of the rescue group.
radio_frequencies = 89200 91400 93600 96200 97800 99400 101600 103800 105400 107000
radio_note_seconds = 12           # real seconds a received radio line stays on screen above the player (0 = the game default only)
radio_line_seconds = 7           # real seconds between the lines of one broadcast (0 = all at once)
radio_range_limit = 0            # 1 = a receiver also needs to be within its own range of the event site
unknown_radio_range = 2000      # tiles; used when a device does not report its range

# SOUNDS (real seconds).  Airdrop: a plane flies over (heard within plane_range of the plane), the crate
# lands plane_drop_seconds later.  Helicopter: heard within heli_range of the helicopter while it comes
# in, then heli_flight_seconds later it crashes: impact + explosion heard within crash_range.
# The ranges can only be LOWERED here; the maximum is distanceMax in media/scripts/RandomEvents_sounds.txt.
sound_enabled = 1
plane_range = 3000
heli_range = 1000
crash_range = 3000
# Sound scripts (names, space separated).  Defaults are the game's own recordings; the mod's own files are
# RE_HeliFlyby / RE_PlaneFlyby / RE_CrashExplosion (deep boom that carries far).
heli_sound = Helicopter
crash_sounds = VehicleCrash PipeBombExplode RE_CrashExplosion
plane_drop_seconds = 20
heli_flight_seconds = 25
heli_hazmat = 1                 # zombies in chemical protection suits near the wreck
wreck_days = 0                  # the wreck stays this many days (0 = until an admin removes it by hand)
message_variants = 1            # 1 = radio texts are picked at random from several variants, 0 = always the first
heli_cargo = 1                  # 1 = the helicopter wreck itself is a lootable container (loot from heli_loot)
heli_crates = 2
heli_crew = 6-11                # zombie crew at the wreck
heli_loot = random              # random  or  set:NAME  (default loot of the helicopter and its crates)
# Own loot for each part (the admin panel, tab Loot, changes them too).  "heli" = take heli_loot.
heli_cargo_loot = heli          # the helicopter itself
heli_crate1_loot = heli         # first crate
heli_crate2_loot = heli         # second crate and every further one

horde_size = 25-49              # only used by the instant "spawn horde" admin command

# HORDE: radio reports zombies coming out of the forests toward a city; the horde arrives in waves.
# horde_waves = zombies per wave, up to 5 numbers, in order.  Everything can also be typed in the panel (tab Horde).
# The first report reaches every receiver; each wave report is broadcast on the wave of the horde and is heard only by

# ONLY in a forest, and the zombies are lured toward the target for horde_attract_minutes.

horde_spread = 30                # radius (tiles) around a group's place over which its zombies are scattered (1-3 per tile)
horde_target_spread = 30         # tiles: zombies are lured to random points this far around the target

#  An item line:   Module.ItemType  [amount]  [chance]  [w=weight]
#       Base.Axe                  -> exactly 1
#       Base.Bandage 2-4          -> 2 to 4
#       Base.Pills x2             -> exactly 2
#       Base.Antibiotics 50%      -> 50 percent chance to be included
#       Base.Nails 5-10 80%       -> 80 percent chance, then 5 to 10
#  A POOL is a bag of items used by "random N from POOL" (w=3 = three times as likely).
#  A SET is the content of one crate.  Inside a set you can use:
#       <item line>                                       always (or with its chance)
#       pick N: A 1-2, B, C 2-3                           choose N different entries of this list (N may be a range: pick 3-7: ...)
#       random N from POOLNAME                            N weighted random draws from a pool (also a range: random 1-5 from bats)

#       guns N-M: Base.Pistol = Base.Bullets9mmBox 3-5 + Base.9mmClip 1-2, Base.X = ...   N..M different weapons, each with ITS OWN ammunition
